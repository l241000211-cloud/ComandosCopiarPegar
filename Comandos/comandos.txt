mkdir practica1 — Crea un directorio o carpeta llamado practica1.  

cd practica1 — Cambia la ubicación actual en la terminal y entra a la carpeta practica1.  

sudo gedit Readme.txt — Abre el editor de texto gráfico con permisos de administrador para redactar y guardar el archivo Readme.txt.  

ls -l — Despliega la lista de elementos en formato largo detallando permisos, propietario, tamaño y fecha de modificación.  

cp ~/Documents/practica1/Readme.txt ~/Documents/practica2/ — Copia el archivo Readme.txt desde su ruta de origen hacia la carpeta de destino practica2.  

cd .. — Retrocede un nivel en la jerarquía del árbol de carpetas para regresar a la carpeta superior o padre.

cp -r carpeta_origen carpeta_destino — Copia una carpeta completa de manera recursiva incluyendo todos los archivos y subcarpetas que contiene.  

rm Readme.txt — Borra de forma inmediata y definitiva un archivo individual del sistema.  

rm -r info — Elimina de forma recursiva una carpeta completa junto con todos los elementos de su interior de manera permanente y sin pasar por la papelera.
